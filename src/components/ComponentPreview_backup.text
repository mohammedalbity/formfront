// Listen for component updates to force re-render
const handleComponentUpdate = (event?: Event) => {
  console.log('🔄 ComponentPreview: Force update triggered', event?.type || 'manual')

  // Enhanced handling for different event types
  if (event) {
    const customEvent = event as CustomEvent

    // Handle table-specific events
    if (event.type === 'table-row-added' || event.type === 'table-updated') {
      console.log('🔄 Table event details:', customEvent.detail)
      if (props.component.type === 'table') {
        console.log('📋 Forcing table component refresh for row/table update')
        forceUpdateKey.value += 3 // Higher increment for table structure changes
      }
    }

    // Handle table cell updates
    else if (event.type === 'table-cell-updated') {
      console.log('📋 Table cell updated event details:', customEvent.detail)
      if (props.component.type === 'table') {
        console.log('📋 Forcing table component refresh for cell update')
        forceUpdateKey.value += 2 // Medium increment for cell updates
      }
    }

    // Handle force table rerender
    else if (event.type === 'force-table-rerender') {
      console.log('🔥 Force table rerender event:', customEvent.detail)
      if (props.component.type === 'table') {
        console.log('🔥 Forcing immediate table rerender')
        forceUpdateKey.value += 5 // Maximum increment for forced rerender

        // Additional delayed update for complex reactivity
        nextTick(() => {
          forceUpdateKey.value += 1
        })
      }
    }

    // Handle tabs events
    else if (event.type === 'tab-added' || event.type === 'tab-updated' || event.type === 'tab-deleted') {
      console.log('📑 Tabs event details:', customEvent.detail)
      if (props.component.type === 'tabs') {
        console.log('📑 Forcing tabs component refresh for tab operation')
        forceUpdateKey.value += 3 // Higher increment for tabs structure changes
      }
    }

    // Handle component-updated events with table and tabs actions
    else if (event.type === 'component-updated') {
      if (customEvent.detail?.action === 'add-table-cell-component' ||
          customEvent.detail?.action === 'add-table-row' ||
          customEvent.detail?.action === 'update-table-header' ||
          customEvent.detail?.action === 'component-duplicated' ||
          customEvent.detail?.action === 'tab-added') {
        console.log('📋 Component updated for table/tabs action:', customEvent.detail)

        if (props.component.type === 'table') {
          console.log('📋 Forcing table component refresh for component addition/duplication')
          // Extra force update for table components
          forceUpdateKey.value += 3

          // Special handling for table header updates
          if (customEvent.detail?.action === 'update-table-header') {
            console.log('📋 Table header updated - forcing immediate refresh')
            forceUpdateKey.value += 2

            // Additional immediate update for header changes
            nextTick(() => {
              forceUpdateKey.value += 1
              console.log('📋 Table header refresh completed')
            })
          }

          // Special handling for table duplication
          if (customEvent.detail?.action === 'component-duplicated' &&
              customEvent.detail?.componentType === 'table') {
            console.log('📋 Table component duplicated - forcing additional refresh')
            forceUpdateKey.value += 2

            // Additional delayed update for table duplication
            setTimeout(() => {
              forceUpdateKey.value += 1
              console.log('📋 Delayed table duplication refresh completed')
            }, 150)
          }
        }

        // Special handling for tabs actions
        if (props.component.type === 'tabs' && customEvent.detail?.action === 'tab-added') {
          console.log('📑 Forcing tabs component refresh for tab addition')
          forceUpdateKey.value += 3
        }
      }
    }
  }

  // If it's a tree-updated event, log additional details
  if (event && event.type === 'tree-updated') {
    const customEvent = event as CustomEvent
    console.log('🌳 Tree updated event details:', customEvent.detail)
    // Force refresh tree data by clearing any cached references
    if (props.component.type === 'tree') {
      console.log('🌳 Forcing tree component refresh')
    }
  }

  forceUpdateKey.value++
  console.log('🔄 ComponentPreview: forceUpdateKey updated to', forceUpdateKey.value)

  // Force Vue to re-render by triggering nextTick
  nextTick(() => {
    console.log('🔄 ComponentPreview: nextTick completed, component should be re-rendered')

    // Additional force update for table components
    if (props.component.type === 'table') {
      // Trigger another update cycle for table components
      setTimeout(() => {
        forceUpdateKey.value++
        console.log('📋 Additional table force update:', forceUpdateKey.value)
      }, 50)
    }
  })
}

/**
 * Get table headers - supports both old and new structure
 */
const getTableHeaders = (tableComponent: any): string[] => {
  try {
    // New structure: headers array
    if (tableComponent.headers && Array.isArray(tableComponent.headers) && tableComponent.headers.length > 0) {
      return tableComponent.headers
    }

    // Old structure: header1, header2, etc. properties
    const oldHeaders = []
    let headerIndex = 1
    while (tableComponent[`header${headerIndex}`]) {
      oldHeaders.push(tableComponent[`header${headerIndex}`])
      headerIndex++
    }

    if (oldHeaders.length > 0) {
      return oldHeaders
    }

    // Fallback: determine from existing rows structure
    if (tableComponent.rows && Array.isArray(tableComponent.rows) && tableComponent.rows.length > 0) {
      const firstRow = tableComponent.rows[0]

      // New structure: row.cells array
      if (firstRow.cells && Array.isArray(firstRow.cells)) {
        const columnCount = firstRow.cells.length
        return Array(columnCount).fill(null).map((_, index) => {
          const columnKey = `table.column${index + 1}`
          return t(columnKey) || `${t('table.columnPrefix') || 'عمود'} ${index + 1}`
        })
      }

      // Old structure: row is array of cells
      if (Array.isArray(firstRow)) {
        const columnCount = firstRow.length
        return Array(columnCount).fill(null).map((_, index) => {
          const columnKey = `table.column${index + 1}`
          return t(columnKey) || `${t('table.columnPrefix') || 'عمود'} ${index + 1}`
        })
      }
    }

    // Default fallback
    return [
      t('table.column1') || `${t('table.columnPrefix') || 'عمود'} 1`,
      t('table.column2') || `${t('table.columnPrefix') || 'عمود'} 2`
    ]
  } catch (error) {
    console.error('Error getting table headers:', error)
    return [
      t('table.column1') || `${t('table.columnPrefix') || 'عمود'} 1`,
      t('table.column2') || `${t('table.columnPrefix') || 'عمود'} 2`
    ]
  }
}

/**
 * Get table rows - supports both old and new structure
 */
const getTableRows = (tableComponent: any): any[] => {
  try {
    console.log('🔍 getTableRows called for table:', {
      hasRows: !!tableComponent.rows,
      isArray: Array.isArray(tableComponent.rows),
      rowsLength: tableComponent.rows?.length || 0,
      rows: tableComponent.rows
    })

    // Initialize default structure if needed
    if (!tableComponent.rows || !Array.isArray(tableComponent.rows)) {
      const headers = getTableHeaders(tableComponent)
      console.log('⚠️ Creating default table rows because no rows exist')
      return createDefaultTableRows(headers.length)
    }

    const headers = getTableHeaders(tableComponent)

    return tableComponent.rows.map((row: any, rowIndex: number) => {
      // New structure: row has cells array
      if (row.cells && Array.isArray(row.cells)) {
        // Ensure row has correct number of cells
        while (row.cells.length < headers.length) {
          row.cells.push(createEmptyTableCell())
        }
        return row
      }

      // Old structure: row is array of cells directly
      if (Array.isArray(row)) {
        const newRow = {
          cells: []
        }

        // Convert old structure to new structure
        for (let i = 0; i < headers.length; i++) {
          if (i < row.length && row[i]) {
            // Convert old cell structure to new
            if (row[i].components && Array.isArray(row[i].components)) {
              (newRow.cells as any[]).push({
                components: row[i].components
              })
            } else {
              (newRow.cells as any[]).push(createEmptyTableCell())
            }
          } else {
            (newRow.cells as any[]).push(createEmptyTableCell())
          }
        }

        return newRow
      }

      // Fallback: create default row
      return createDefaultTableRow(headers.length, rowIndex)
    })
  } catch (error) {
    console.error('Error getting table rows:', error)
    const headers = getTableHeaders(tableComponent)
    return createDefaultTableRows(headers.length)
  }
}

/**
 * Get table cells for a specific row
 */
const getTableCells = (row: any, tableComponent: any): any[] => {
  try {
    console.log('🔍 getTableCells called for row:', {
      hasRow: !!row,
      hasCells: !!row?.cells,
      isArray: Array.isArray(row?.cells),
      cellsLength: row?.cells?.length || 0,
      cells: row?.cells
    })

    if (!row || !row.cells || !Array.isArray(row.cells)) {
      const headers = getTableHeaders(tableComponent)
      console.log('⚠️ Creating empty cells because row has no cells')
      return Array(headers.length).fill(null).map(() => createEmptyTableCell())
    }

    const headers = getTableHeaders(tableComponent)
    const cells = [...row.cells]

    console.log('🔍 Processing table cells:', {
      headersLength: headers.length,
      cellsLength: cells.length,
      cellsData: cells.map((cell, index) => ({
        index,
        hasComponents: !!cell?.components,
        componentsLength: cell?.components?.length || 0,
        components: cell?.components
      }))
    })

    // Ensure we have the right number of cells
    while (cells.length < headers.length) {
      cells.push(createEmptyTableCell())
    }

    return cells.slice(0, headers.length)
  } catch (error) {
    console.error('Error getting table cells:', error)
    const headers = getTableHeaders(tableComponent)
    return Array(headers.length).fill(null).map(() => createEmptyTableCell())
  }
}

const validateField = (gridComponent: any, value: any): boolean => {
  if (!gridComponent.validate) return true

  // Required validation
  if (gridComponent.validate.required) {
    if (value === null || value === undefined || value === '') {
      return false
    }
  }

  // Email validation
  if (gridComponent.type === 'email' && value) {
    const emailRegex = /^[^\s@]+@[^\s@]+\.[^\s@]+$/
    if (!emailRegex.test(value)) {
      return false
    }
  }

  // URL validation
  if (gridComponent.type === 'url' && value) {
    try {
      new URL(value)
    } catch {
      return false
    }
  }

  // Min/Max length validation
  if (gridComponent.validate.minLength && value && value.length < gridComponent.validate.minLength) {
    return false
  }

  if (gridComponent.validate.maxLength && value && value.length > gridComponent.validate.maxLength) {
    return false
  }

  return true
}

// Drag and Drop handlers for Panel - Optimized
const handlePanelDragOver = (event: DragEvent) => {
  event.preventDefault()
  if (event.dataTransfer) {
    event.dataTransfer.dropEffect = 'copy'
  }
  isDragOverPanel.value = true
}

const handlePanelDragEnter = (event: DragEvent) => {
  event.preventDefault()
  isDragOverPanel.value = true
}

const handlePanelDragLeave = (event: DragEvent) => {
  event.preventDefault()
  const currentTarget = event.currentTarget as Element
  const relatedTarget = event.relatedTarget as Element

  if (currentTarget && !currentTarget.contains(relatedTarget)) {
    isDragOverPanel.value = false
  }
}

const handlePanelDrop = (event: DragEvent) => {
  event.preventDefault()
  event.stopPropagation()
  isDragOverPanel.value = false

  if (!event.dataTransfer) {
    console.error('❌ No dataTransfer available in panel drop event')
    return
  }

  try {
    const componentDataString = event.dataTransfer.getData('application/json')
    if (!componentDataString) {
      console.error('❌ No component data found in panel drop event')
      return
    }

    const componentData = JSON.parse(componentDataString) as ComponentDefinition

    // Validate component data
    if (!componentData.type) {
      console.error('❌ Invalid component data: missing type')
      return
    }

    // Emit the drop event with container path for panel
    const containerPath = 'components'
    emit('drop-component', componentData, containerPath)

    // Minimal success logging
    console.log('✅ Component dropped in panel:', componentData.type)

  } catch (error) {
    console.error('❌ Error processing panel drop:', error)
  }
}

const dataGridSort = ref({
  column: null as number | null,
  direction: 'asc' as 'asc' | 'desc'
})

const dataGridFilter = ref({
  searchTerm: '',
  columnFilters: {} as Record<number, string>
})

const dataGridPagination = ref({
  currentPage: 1,
  itemsPerPage: 10,
  totalItems: 0
})

// Sort DataGrid by column
const sortDataGrid = (columnIndex: number) => {
  console.log('🔄 Sorting DataGrid by column:', columnIndex)

  if (dataGridSort.value.column === columnIndex) {
    // Toggle direction if same column
    dataGridSort.value.direction = dataGridSort.value.direction === 'asc' ? 'desc' : 'asc'
  } else {
    // Set new column and default to ascending
    dataGridSort.value.column = columnIndex
    dataGridSort.value.direction = 'asc'
  }

  // Apply sorting to the data
  applySortToDataGrid()

  console.log('✅ DataGrid sorted:', dataGridSort.value)
}

// Apply sorting to DataGrid data
const applySortToDataGrid = () => {
  if (!component.value.defaultValue || dataGridSort.value.column === null) return

  const columnKey = component.value.components?.[dataGridSort.value.column]?.key
  if (!columnKey) return

  const sortedData = [...component.value.defaultValue].sort((a, b) => {
    const aValue = a[columnKey] || ''
    const bValue = b[columnKey] || ''

    const comparison = String(aValue).localeCompare(String(bValue))
    return dataGridSort.value.direction === 'asc' ? comparison : -comparison
  })

  // Update the component data
  ;(component.value as any).defaultValue = sortedData
  forceUpdateKey.value++
}

// Filter DataGrid data
const filterDataGrid = (searchTerm: string) => {
  console.log('🔍 Filtering DataGrid with term:', searchTerm)
  dataGridFilter.value.searchTerm = searchTerm
  dataGridPagination.value.currentPage = 1 // Reset to first page
  forceUpdateKey.value++
}

// Get filtered DataGrid data
const getFilteredDataGridData = () => {
  if (!component.value.defaultValue) return []

  let filteredData = [...component.value.defaultValue]

  // Apply search filter
  if (dataGridFilter.value.searchTerm) {
    const searchTerm = dataGridFilter.value.searchTerm.toLowerCase()
    filteredData = filteredData.filter(row => {
      return Object.values(row).some(value =>
        String(value).toLowerCase().includes(searchTerm)
      )
    })
  }

  // Apply column filters
  Object.entries(dataGridFilter.value.columnFilters).forEach(([columnIndex, filterValue]) => {
    if (filterValue) {
      const columnKey = component.value.components?.[parseInt(columnIndex)]?.key
      if (columnKey) {
        filteredData = filteredData.filter(row =>
          String(row[columnKey] || '').toLowerCase().includes(filterValue.toLowerCase())
        )
      }
    }
  })

  return filteredData
}

// Get paginated DataGrid data
const getPaginatedDataGridData = () => {
  const filteredData = getFilteredDataGridData()
  dataGridPagination.value.totalItems = filteredData.length

  const startIndex = (dataGridPagination.value.currentPage - 1) * dataGridPagination.value.itemsPerPage
  const endIndex = startIndex + dataGridPagination.value.itemsPerPage

  return filteredData.slice(startIndex, endIndex)
}

  if (confirm(t('survey.confirmDeleteQuestion'))) {
    emit('delete-survey-question', question, index, component.value)
  }
}

  showSurveyQuestionModal.value = false
  editingSurveyQuestion.value = null
  editingSurveyQuestionIndex.value = null
  isEditingSurveyQuestion.value = false
}

const saveSurveyQuestion = (question: any) => {
  console.log('💾 Saving survey question:', question)

  if (isEditingSurveyQuestion.value && editingSurveyQuestionIndex.value !== null) {
    // Update existing question
    emit('update-survey-question', question, editingSurveyQuestionIndex.value, component.value)
  } else {
    // Add new question
    emit('add-survey-question', question, component.value)
  }

  closeSurveyQuestionModal()
}
